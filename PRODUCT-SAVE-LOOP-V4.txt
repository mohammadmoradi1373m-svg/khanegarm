  const uploadedUrls = [];
  for(const rawFile of pfNewFiles){
    const [file,thumbFile]=await imageUploadVariants(rawFile);
    if(!file){
      showToast(rawFile&&rawFile.size>PRODUCT_IMAGE_MAX_INPUT_BYTES
        ? 'حجم عکس اولیه خیلی زیاد است (حداکثر ۲۵ مگابایت)'
        : 'این فایل تصویر قابل پردازش نیست؛ لطفاً عکس معمولی JPG، PNG یا WEBP انتخاب کنید');
      btn.disabled=false;btn.textContent='ذخیره';return;
    }
    const basePath=`${id}_${Date.now()}_${Math.floor(Math.random()*10000)}`;
    const ext=file.type==='image/webp'?'webp':file.type==='image/png'?'png':'jpg';
    const thumbExt=thumbFile?.type==='image/webp'?'webp':thumbFile?.type==='image/png'?'png':'jpg';
    const path=`${basePath}.${ext}`;
    const thumbPath=`${basePath}_thumb.${thumbExt}`;
    let upErr;
    try{
      const bucket=sbClient.storage.from('product-images');
      const mainRes=await withTimeout(bucket.upload(path,file,{upsert:true,cacheControl:'31536000',contentType:file.type}),20000,'UPLOAD_TIMEOUT');
      upErr=mainRes.error;
      if(!upErr&&thumbFile){
        const thumbRes=await withTimeout(bucket.upload(thumbPath,thumbFile,{upsert:true,cacheControl:'31536000',contentType:thumbFile.type}),20000,'UPLOAD_TIMEOUT');
        // تصویر اصلی نباید به خاطر thumbnail از دست برود.
        if(thumbRes.error&&file.type!=='image/webp')upErr=thumbRes.error;
      }
    }catch(e){
      showToast('آپلود عکس خیلی طول کشید (احتمالاً مشکل اتصال یا فیلترینگ)، دوباره امتحان کن');
      btn.disabled=false;btn.textContent='ذخیره';return;
    }
    if(upErr){showToast('خطا در آپلود عکس: '+describeLoadError(upErr));btn.disabled=false;btn.textContent='ذخیره';return;}
    const {data:pub}=sbClient.storage.from('product-images').getPublicUrl(path);
    uploadedUrls.push(pub.publicUrl);
  }
